import 'package:factory_notification_kit/factory_notification_kit.dart';
import 'package:factory_notification_flutter_adapter/factory_flutter_notification_adapter.dart';
import 'package:factory_notification_flutter_adapter/plugin_android_notification_backend.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('real Android API29 channel scheduling and cancellation', (tester) async {
    await tester.pumpWidget(const MaterialApp(home: Scaffold(body: Text('X10 Android'))));
    const channel = MethodChannel('x10/android_sdk');
    final sdk = await channel.invokeMethod<int>('sdkInt');
    expect(sdk, 29);
    final plugin = FlutterLocalNotificationsPlugin();
    var initialized = false;
    var initCount = 0;
    Future<bool> init() async {
      if (initialized) return true;
      initCount++;
      initialized = await plugin.initialize(
        settings: const InitializationSettings(
          android: AndroidInitializationSettings('@mipmap/ic_launcher'),
        ),
        onDidReceiveNotificationResponse: (_) {},
      ) ?? false;
      return initialized;
    }
    String? nativeScheduleFailure;
    final backend = PluginAndroidNotificationBackend(
      plugin: plugin, nowUtc: () => DateTime.now().toUtc(),
      channelId: 'x10_android_smoke', channelName: 'X10 reminders',
      ensureConsumerPluginInitialized: init,
      androidSdkInt: () => channel.invokeMethod<int>('sdkInt'),
      onNativeOperationError: (operation, error) {
        if (operation == 'schedule') {
          nativeScheduleFailure = error is PlatformException
              ? '${error.code}: ${error.message}'
              : error.runtimeType.toString();
        }
      },
    );
    final state = await backend.probeDevice();
    expect(state.supported, isTrue);
    expect(state.permissionGranted, isTrue);
    expect(state.channelEnabled, isTrue);
    expect(state.schedulerAvailable, isTrue);
    final coordinator = NotificationCoordinator(
      FlutterNotificationPlatformPort(backend),
      nowUtc: () => DateTime.now().toUtc(),
    );
    const stableId = 'x10:android:smoke';
    final outcome = await coordinator.schedule(ReminderRequest(
      stableId: stableId,
      scheduledAtUtc: DateTime.now().toUtc().add(const Duration(minutes: 5)),
      title: 'Android native scheduling test',
      body: 'Registered, not delivered', timezoneId: 'UTC',
    ));
    expect(outcome, ScheduleDecision.acceptedInexact,
        reason: 'Native schedule error: ${nativeScheduleFailure ?? 'none reported'}');
    expect((await plugin.pendingNotificationRequests()).map((p) => p.id),
      contains(notificationIdFor(stableId)));
    expect(await coordinator.cancel(stableId), isTrue);
    expect((await plugin.pendingNotificationRequests()).map((p) => p.id),
      isNot(contains(notificationIdFor(stableId))));
    expect(initCount, 1);
    expect((await FlutterNotificationPlatformPort(backend).probe())
      .deliveryReceiptVerified, isFalse);
  });
}
